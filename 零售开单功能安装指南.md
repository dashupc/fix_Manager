```python
            CREATE TABLE IF NOT EXISTS retail_order_items (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                order_id INTEGER,
                product_name TEXT NOT NULL,
                unit TEXT,
                quantity REAL DEFAULT 1.0,
                unit_price REAL DEFAULT 0.0,
                amount REAL DEFAULT 0.0,
                source TEXT,
                cost REAL DEFAULT 0.0,
                notes TEXT,
                FOREIGN KEY (order_id) REFERENCES retail_orders(id)
            )
        ''')
```

```python
    def setup_retail_tab(self):
        """设置零售开单标签页"""
        # 主框架
        main_frame = ttk.Frame(self.retail_tab, padding="10")
        main_frame.pack(fill="both", expand=True)
        
        # 顶部：客户单位输入
        top_frame = ttk.Frame(main_frame)
        top_frame.pack(fill="x", pady=(0, 10))
        
        ttk.Label(top_frame, text="客户单位:", font=('Arial', 10)).pack(side="left", padx=5)
        self.retail_customer_unit = ttk.Entry(top_frame, width=30)
        self.retail_customer_unit.pack(side="left", padx=5)
        
        # 中间：商品列表表格

        table_frame.pack(fill="both", expand=True, pady=10)

        self.retail_tree = ttk.Treeview(table_frame, columns=columns, show='headings', height=15)
        
        # 设置列标题
        self.retail_tree.heading('product_name', text='货物名称')
        self.retail_tree.heading('unit', text='单位')
        self.retail_tree.heading('quantity', text='数量')

        self.retail_tree.heading('amount', text='金额')
        self.retail_tree.heading('source', text='货物来源')
        self.retail_tree.heading('cost', text='成本')
        self.retail_tree.heading('notes', text='备注')
        
        # 设置列宽
        self.retail_tree.column('product_name', width=120)
        self.retail_tree.column('unit', width=60)
        self.retail_tree.column('quantity', width=60)
        self.retail_tree.column('unit_price', width=80)
        self.retail_tree.column('amount', width=80)
        self.retail_tree.column('source', width=100)
        self.retail_tree.column('cost', width=80)
        self.retail_tree.column('notes', width=150)
        
        # 添加滚动条
        scrollbar = ttk.Scrollbar(table_frame, orient="vertical", command=self.retail_tree.yview)
        self.retail_tree.configure(yscrollcommand=scrollbar.set)
        
        self.retail_tree.pack(side="left", fill="both", expand=True)
        scrollbar.pack(side="right", fill="y")
        
        # 添加行按钮
        add_row_btn = ttk.Button(main_frame, text="添加一行", command=self.add_retail_row)
        add_row_btn.pack(pady=5)
        
        # 底部：总价和利润显示
        summary_frame = ttk.Frame(main_frame)
        summary_frame.pack(fill="x", pady=10)
        
        ttk.Label(summary_frame, text="总价:", font=('Arial', 11, 'bold')).pack(side="left", padx=10)
        self.retail_total_label = ttk.Label(summary_frame, text="0.00", font=('Arial', 11))
        self.retail_total_label.pack(side="left", padx=5)
        
        ttk.Label(summary_frame, text="利润:", font=('Arial', 11, 'bold')).pack(side="left", padx=10)
        self.retail_profit_label = ttk.Label(summary_frame, text="0.00", font=('Arial', 11))
        self.retail_profit_label.pack(side="left", padx=5)
        
        # 按钮框架
        btn_frame = ttk.Frame(main_frame)
        btn_frame.pack(fill="x", pady=10)
        
        ttk.Button(btn_frame, text="保存", command=self.save_retail_order).pack(side="left", padx=5)
        ttk.Button(btn_frame, text="取消", command=self.cancel_retail_order).pack(side="left", padx=5)
        ttk.Button(btn_frame, text="删除选中行", command=self.delete_retail_row).pack(side="left", padx=5)
        
        # 双击编辑

    def add_retail_row(self):
        """添加新的零售商品行"""
        dialog = tk.Toplevel(self)
        dialog.title("添加商品")
        dialog.geometry("500x400")
        self._set_window_icon(dialog)
        self.center_window_manual(dialog, 500, 400)
        
        frame = ttk.Frame(dialog, padding="20")
        frame.pack(fill="both", expand=True)
        
        # 输入字段
        fields = [
            ("货物名称:", "product_name"),
            ("单位:", "unit"),
            ("数量:", "quantity"),
            ("单价:", "unit_price"),
            ("货物来源:", "source"),
            ("成本:", "cost"),
            ("备注:", "notes")
        ]
        
        entries = {}
        for i, (label, key) in enumerate(fields):
            ttk.Label(frame, text=label).grid(row=i, column=0, sticky="w", pady=5, padx=5)
            
            if key == "source":
                # 使用下拉框显示货物来源
                sources = self.db.fetch_all("SELECT DISTINCT config_value FROM config_data WHERE config_type='part_source'")
                source_values = [s[0] for s in sources] if sources else ['自购', '客户提供', '其他']
                entry = ttk.Combobox(frame, values=source_values, width=30)
                if source_values:
                    entry.set(source_values[0])
            else:
                entry = ttk.Entry(frame, width=30)
            
            entry.grid(row=i, column=1, pady=5, padx=5)
            entries[key] = entry
        
        def save_item():
            try:
                product_name = entries['product_name'].get().strip()
                if not product_name:
                    messagebox.showwarning("警告", "货物名称不能为空！")
                    return
                
                unit = entries['unit'].get().strip()
                quantity = float(entries['quantity'].get() or 0)
                unit_price = float(entries['unit_price'].get() or 0)
                amount = quantity * unit_price
                source = entries['source'].get().strip()
                cost = float(entries['cost'].get() or 0)
                notes = entries['notes'].get().strip()
                
                # 添加到表格
                self.retail_tree.insert('', 'end', values=(
                    product_name, unit, quantity, unit_price, amount, source, cost, notes
                ))
                
                # 更新总计
                self.update_retail_totals()
                
                dialog.destroy()
            except ValueError as e:
                messagebox.showerror("错误", f"数据格式错误: {e}")
        
        # 按钮
        btn_frame = ttk.Frame(frame)
        btn_frame.grid(row=len(fields), column=0, columnspan=2, pady=20)
        
        ttk.Button(btn_frame, text="确定", command=save_item).pack(side="left", padx=5)
        ttk.Button(btn_frame, text="取消", command=dialog.destroy).pack(side="left", padx=5)
    
    def edit_retail_item(self, event):
        """编辑零售商品行"""
        selection = self.retail_tree.selection()
        if not selection:
            return
        
        item = selection[0]
        values = self.retail_tree.item(item, 'values')
        
        dialog = tk.Toplevel(self)
        dialog.title("编辑商品")
        dialog.geometry("500x400")
        self._set_window_icon(dialog)
        self.center_window_manual(dialog, 500, 400)
        
        frame = ttk.Frame(dialog, padding="20")
        frame.pack(fill="both", expand=True)
        
        # 输入字段
        fields = [
            ("货物名称:", "product_name", values[0]),
            ("单位:", "unit", values[1]),
            ("数量:", "quantity", values[2]),
            ("单价:", "unit_price", values[3]),
            ("货物来源:", "source", values[5]),
            ("成本:", "cost", values[6]),
            ("备注:", "notes", values[7])
        ]
        
        entries = {}
        for i, (label, key, default_value) in enumerate(fields):
            ttk.Label(frame, text=label).grid(row=i, column=0, sticky="w", pady=5, padx=5)
            
            if key == "source":
                sources = self.db.fetch_all("SELECT DISTINCT config_value FROM config_data WHERE config_type='part_source'")
                source_values = [s[0] for s in sources] if sources else ['自购', '客户提供', '其他']
                entry = ttk.Combobox(frame, values=source_values, width=30)
                entry.set(default_value)
            else:
                entry = ttk.Entry(frame, width=30)
                entry.insert(0, default_value)
            
            entry.grid(row=i, column=1, pady=5, padx=5)
            entries[key] = entry
        
        def save_item():
            try:
                product_name = entries['product_name'].get().strip()
                if not product_name:
                    messagebox.showwarning("警告", "货物名称不能为空！")
                    return
                
                unit = entries['unit'].get().strip()
                quantity = float(entries['quantity'].get() or 0)
                unit_price = float(entries['unit_price'].get() or 0)
                amount = quantity * unit_price
                source = entries['source'].get().strip()
                cost = float(entries['cost'].get() or 0)
                notes = entries['notes'].get().strip()
                
                # 更新表格
                self.retail_tree.item(item, values=(
                    product_name, unit, quantity, unit_price, amount, source, cost, notes
                ))
                
                # 更新总计
                self.update_retail_totals()
                
                dialog.destroy()
            except ValueError as e:
                messagebox.showerror("错误", f"数据格式错误: {e}")
        
        # 按钮
        btn_frame = ttk.Frame(frame)
        btn_frame.grid(row=len(fields), column=0, columnspan=2, pady=20)
        
        ttk.Button(btn_frame, text="确定", command=save_item).pack(side="left", padx=5)
        ttk.Button(btn_frame, text="取消", command=dialog.destroy).pack(side="left", padx=5)
    
    def delete_retail_row(self):
        """删除选中的零售商品行"""
        selection = self.retail_tree.selection()
        if not selection:
            messagebox.showwarning("警告", "请先选择要删除的行！")
            return
        
        if messagebox.askyesno("确认", "确定要删除选中的行吗？"):
            for item in selection:
                self.retail_tree.delete(item)
            self.update_retail_totals()
    
    def update_retail_totals(self):
        """更新零售订单的总价和利润"""
        total_amount = 0.0
        total_cost = 0.0
        
        for item in self.retail_tree.get_children():
            values = self.retail_tree.item(item, 'values')
            amount = float(values[4]) if values[4] else 0.0
            cost = float(values[6]) if values[6] else 0.0
            
            total_amount += amount
            total_cost += cost
        
        profit = total_amount - total_cost
        
        self.retail_total_label.config(text=f"{total_amount:.2f}")
        self.retail_profit_label.config(text=f"{profit:.2f}")
    
    def save_retail_order(self):
        """保存零售订单"""
        customer_unit = self.retail_customer_unit.get().strip()
        
        # 检查是否有商品
        items = self.retail_tree.get_children()
        if not items:
            messagebox.showwarning("警告", "请至少添加一个商品！")
            return
        
        # 计算总价和利润
        total_amount = 0.0
        total_cost = 0.0
        
        for item in items:
            values = self.retail_tree.item(item, 'values')
            total_amount += float(values[4]) if values[4] else 0.0
            total_cost += float(values[6]) if values[6] else 0.0
        
        profit = total_amount - total_cost
        
        try:
            # 保存订单主表
            order_id = self.db.execute_query(
                '''INSERT INTO retail_orders (customer_unit, total_amount, total_cost, profit, created_at)
                   VALUES (?, ?, ?, ?, ?)''',
                (customer_unit, total_amount, total_cost, profit, datetime.now().strftime('%Y-%m-%d %H:%M:%S'))
            )
            
            # 保存订单明细
            for item in items:
                values = self.retail_tree.item(item, 'values')
                self.db.execute_query(
                    '''INSERT INTO retail_order_items 
                       (order_id, product_name, unit, quantity, unit_price, amount, source, cost, notes)
                       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)''',
                    (order_id, values[0], values[1], values[2], values[3], values[4], values[5], values[6], values[7])
                )
            
            messagebox.showinfo("成功", f"零售订单已保存！\n订单号: {order_id}")
            self.cancel_retail_order()
            
        except Exception as e:
            messagebox.showerror("错误", f"保存失败: {e}")
    
    def cancel_retail_order(self):
        """取消/清空零售订单"""
        self.retail_customer_unit.delete(0, tk.END)
        for item in self.retail_tree.get_children():
            self.retail_tree.delete(item)
        self.update_retail_totals()
```
